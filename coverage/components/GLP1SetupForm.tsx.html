import React, { useMemo, useState } from "react";
import { GLP1MedicationProfile, MedicationDosePeriod } from "../types";
 
type Props = { profile: GLP1MedicationProfile | null; dose: MedicationDosePeriod | null; onSave: (p: Omit<GLP1MedicationProfile,"id"|"createdAt"|"updatedAt">, d: Pick<MedicationDosePeriod,"doseValue"|"doseUnit"|"prescribedDoseLabel"|"effectiveStartDate"|"changeNote">) => void; onCancel?: () => void };
const DAYS = ["Sun","Mon","Tue","Wed","Thu","Fri","Sat"];
export const GLP1SetupForm: React.FC<Props> = ({ profile, dose, onSave, onCancel }) => {
  const [step, setStep] = useState<1|2>(1); const today = new Date().toISOString().slice(0,10);
  const [name,setName]=useState(profile?.medicationName||""); const [form,setForm]=useState(profile?.medicationForm||"weekly_injection"); const [schedule,setSchedule]=useState(profile?.scheduleType||"weekly");
  const [days,setDays]=useState<number[]>(profile?.scheduledDays||[new Date().getDay()]); const [time,setTime]=useState(profile?.scheduledTime||"08:00"); const [start,setStart]=useState(profile?.treatmentStartDate||today);
  const [doseLabel,setDoseLabel]=useState(dose?.prescribedDoseLabel||""); const [doseValue,setDoseValue]=useState<string>(dose?.doseValue?.toString()||""); const [unit,setUnit]=useState(dose?.doseUnit||"mg"); const [instructions,setInstructions]=useState(profile?.userEnteredInstructions||"");
  const [trackSite,setTrackSite]=useState(profile?.trackInjectionSite||false); const [reminders,setReminders]=useState(profile?.remindersEnabled||false); const [changeNote,setChangeNote]=useState("");
  const normalizedDays=useMemo(()=>schedule==="daily"?[0,1,2,3,4,5,6]:days,[schedule,days]); const valid=!!(name.trim()&&doseLabel.trim()&&start&&time&&normalizedDays.length);
  const input="w-full px-3 py-2.5 text-sm bg-slate-50 border border-slate-200 rounded-xl focus:ring-2 focus:ring-violet-500 outline-none";
  const submit=()=>onSave({ medicationName:name.trim(), medicationForm:form as any, scheduleType:schedule as any, scheduledDays:normalizedDays, scheduledTime:time, treatmentStartDate:start, userEnteredInstructions:instructions.trim()||null, remindersEnabled:reminders, trackInjectionSite:trackSite, isActive:true }, { doseValue:doseValue?Number(doseValue):null, doseUnit:unit||null, prescribedDoseLabel:doseLabel.trim(), effectiveStartDate:start, changeNote:changeNote.trim()||null });
  if(step===2) return <div className="space-y-4"><h3 className="text-lg font-extrabold">Review GLP-1 setup</h3><dl className="grid grid-cols-2 gap-3 text-sm bg-slate-50 p-4 rounded-2xl"><div><dt className="text-slate-500">Medication</dt><dd className="font-bold">{name}</dd></div><div><dt className="text-slate-500">Prescribed dose</dt><dd className="font-bold">{doseLabel}</dd></div><div><dt className="text-slate-500">Schedule</dt><dd className="font-bold">{schedule}, {time}</dd></div><div><dt className="text-slate-500">Start date</dt><dd className="font-bold">{start}</dd></div><div><dt className="text-slate-500">Reminders</dt><dd className="font-bold">{reminders?"Preference saved":"Off"}</dd></div></dl><p className="p-3 bg-amber-50 border border-amber-200 text-xs text-amber-900 rounded-xl">Follow the instructions from your prescriber and medication guide. This app records information but does not determine your dose or tell you when to change it.</p><div className="flex justify-end gap-2"><button onClick={()=>setStep(1)} className="px-4 py-2 rounded-xl bg-slate-100 text-sm">Back</button><button onClick={submit} className="px-4 py-2 rounded-xl bg-violet-600 text-white text-sm font-bold">Confirm setup</button></div></div>;
  return <form onSubmit={e=>{e.preventDefault();if(valid)setStep(2)}} className="space-y-4"><h3 className="text-lg font-extrabold">{profile?"GLP-1 settings":"Set up GLP-1 tracking"}</h3><div className="grid sm:grid-cols-2 gap-3"><label className="text-xs font-semibold">Medication name<input required value={name} onChange={e=>setName(e.target.value)} className={input}/></label><label className="text-xs font-semibold">Medication form<select value={form} onChange={e=>setForm(e.target.value as any)} className={input}><option value="weekly_injection">Weekly injection</option><option value="daily_injection">Daily injection</option><option value="daily_tablet">Daily tablet</option><option value="other">Other prescribed form</option></select></label><label className="text-xs font-semibold">Prescribed dose label<input required placeholder="e.g. 2.5 mg" value={doseLabel} onChange={e=>setDoseLabel(e.target.value)} className={input}/></label><div className="grid grid-cols-2 gap-2"><label className="text-xs font-semibold">Numeric dose<input type="number" min="0" step="any" value={doseValue} onChange={e=>setDoseValue(e.target.value)} className={input}/></label><label className="text-xs font-semibold">Unit<input value={unit} onChange={e=>setUnit(e.target.value)} className={input}/></label></div><label className="text-xs font-semibold">Schedule<select value={schedule} onChange={e=>setSchedule(e.target.value as any)} className={input}><option value="daily">Daily</option><option value="weekly">Weekly</option><option value="custom">Selected weekdays</option></select></label><label className="text-xs font-semibold">Planned time<input type="time" required value={time} onChange={e=>setTime(e.target.value)} className={input}/></label><label className="text-xs font-semibold">Treatment / dose effective date<input type="date" required value={start} onChange={e=>setStart(e.target.value)} className={input}/></label><label className="text-xs font-semibold">Dose-change note<input value={changeNote} onChange={e=>setChangeNote(e.target.value)} className={input}/></label></div>{schedule!=="daily"&&<fieldset><legend className="text-xs font-semibold mb-1">Planned weekday{schedule==="custom"?"s":""}</legend><div className="flex flex-wrap gap-1">{DAYS.map((d,i)=><button type="button" key={d} aria-pressed={days.includes(i)} onClick={()=>setDays(schedule==="weekly"?[i]:days.includes(i)?days.filter(x=>x!==i):[...days,i])} className={`px-3 py-2 rounded-lg text-xs ${days.includes(i)?"bg-violet-600 text-white":"bg-slate-100"}`}>{d}</button>)}</div></fieldset>}<label className="text-xs font-semibold block">Prescriber-entered instructions<textarea value={instructions} onChange={e=>setInstructions(e.target.value)} className={input}/></label><div className="flex flex-wrap gap-4 text-sm"><label><input type="checkbox" checked={trackSite} onChange={e=>setTrackSite(e.target.checked)}/> Track injection site</label><label><input type="checkbox" checked={reminders} onChange={e=>setReminders(e.target.checked)}/> Save reminder preference</label></div><p className="text-xs text-slate-500">Reminder preferences are stored for future use; this version does not send notifications.</p><div className="flex justify-end gap-2">{onCancel&&<button type="button" onClick={onCancel} className="px-4 py-2 rounded-xl bg-slate-100 text-sm">Cancel</button>}<button disabled={!valid} className="px-4 py-2 rounded-xl bg-violet-600 disabled:opacity-50 text-white text-sm font-bold">Review setup</button></div></form>;
};
